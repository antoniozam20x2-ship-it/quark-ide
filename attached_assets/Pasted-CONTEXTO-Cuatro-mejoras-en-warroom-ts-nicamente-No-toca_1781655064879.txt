FIX 2 — Conectar reporte de Sniper OS
═══════════════════════════════════════

1. Crear función fetchSniperOSReport():

async function fetchSniperOSReport(): 
  Promise<string | null> {
  try {
    const res = await fetch(
      'https://workspacesniper-os-production.up.railway.app
       /api/oracle-report',
      { signal: AbortSignal.timeout(5000) }
    )
    const data = await res.json()
    if (!data.summary) return null
    
    // Construir texto enriquecido con todos los campos
    let report = data.summary
    
    if (data.risk_level) {
      report += `\n\nNivel de riesgo actual: ${data.risk_level.toUpperCase()}`
    }
    if (data.recomendacion) {
      report += `\nAcción recomendada: ${data.recomendacion}`
    }
    if (data.fortalezas?.length) {
      report += `\nFortalezas: ${data.fortalezas.join(', ')}`
    }
    if (data.debilidades?.length) {
      report += `\nDebilidades: ${data.debilidades.join(', ')}`
    }
    
    return report
  } catch {
    return null
  }
}
