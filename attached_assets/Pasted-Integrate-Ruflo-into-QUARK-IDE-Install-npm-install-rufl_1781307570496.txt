  key: string,

  namespace: string = 'quark-ide'
): Promise<void>

export async function searchMemory(
  query: string,

→ Falls back to simple keyword search on memory.json

INTEGRATION 2 — Memory API Routes
In backend/src/routes/memory.ts (new file):

POST /api/memory/save
Body: { key, content, namespace? }
→ Saves to ruflo memory
→ Returns: { saved: true, key }

GET /api/memory/search?q=query&ns=namespace
→ Searches memory
→ Returns: { results: string[], count: number }

POST /api/memory/project
Body: { projectName, files: [{name, content}] }
→ Saves full project
→ Returns: { saved: true, filesCount: number }

GET /api/memory/list
→ Lists all saved namespaces and keys
→ Returns: { namespaces: string[], count: number }

Register routes in backend/src/index.ts:
import memoryRoutes from './routes/memory';
app.use('/api/memory', memoryRoutes);

INTEGRATION 5 — Ruflo Swarm for War Room BOARD
In backend/src/routes/warroom.ts:
Add POST /api/warroom/swarm endpoint:

Body: { challenge: string }

This endpoint calls all 4 board members in PARALLEL
using Promise.all() instead of sequential calls:

const [ceo, cto, designer, qa] = await Promise.all([
  callBoardMember('CEO', challenge),
  callBoardMember('CTO', challenge),  
  callBoardMember('Designer', challenge),
  callBoardMember('QA', challenge)
]);

Then generates consensus from all 4 responses:
const consensus = await generateConsensus(
  challenge, { ceo, cto, designer, qa }
);

Return: { ceo, cto, designer, qa, consensus, 
          processingTime: ms }

Update frontend BoardRoom to use /api/warroom/swarm
Show "⚡ Parallel Mode" badge when using swarm.
