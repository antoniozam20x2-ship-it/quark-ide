1. Crear función fetchSignalOSReport():
   
   async function fetchSignalOSReport(): 
     Promise<string | null> {
     try {
       const res = await fetch(
         'https://workspaceapi-server-production-f248.up.railway.app
          /api/stats/daily-report',
         { signal: AbortSignal.timeout(5000) }
       )
       const data = await res.json()
       if (data.status === 'no_report' || !data.report) 
         return null
       return data.report  // texto plano con emojis
     } catch {
       return null  // nunca bloquear el swarm si falla
     }
   }
