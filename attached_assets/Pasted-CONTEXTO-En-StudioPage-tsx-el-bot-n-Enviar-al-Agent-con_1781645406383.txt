1. Agregar dos estados nuevos:
   const [studioHtml, setStudioHtml] = useState<string>('')
   const [buildLoading, setBuildLoading] = useState(false)

2. Crear función buildAndPreview():
   
   async function buildAndPreview() {
     if (!engineeredPrompt) return
     setBuildLoading(true)
     
     try {
       // Primero genera el código completo con /generate
       const genRes = await fetch('/api/agent/generate', {
         method: 'POST',
         headers: { 'Content-Type': 'application/json' },
         body: JSON.stringify({
           prompt: engineeredPrompt,
           repo: 'none'  // sin repo — creación libre
         })
       })
       const genData = await genRes.json()
       const generatedCode = genData.code || genData.result || ''
       
       // Luego convierte a HTML ejecutable
       const htmlRes = await fetch('/api/agent/generate-html', {
         method: 'POST',
         headers: { 'Content-Type': 'application/json' },
         body: JSON.stringify({ code: generatedCode })
       })
       const htmlData = await htmlRes.json()
       const html = htmlData.html || htmlData.result || ''
       
       setStudioHtml(html)
     } catch (err) {
       console.error('buildAndPreview error:', err)
     } finally {
       setBuildLoading(false)
     }
   }

3. Reemplazar el botón actual:
   
   DE:
   <button onClick={() => onSendToAgent(engineeredPrompt || brief)}>
     ⚡ Enviar al Agent → construir
   </button>
   
   A:
   <button 
     onClick={buildAndPreview}
     disabled={buildLoading || !engineeredPrompt}
   >
     {buildLoading ? '⚡ Construyendo...' : '⚡ Construir y Previsualizar'}
   </button>
